cmake_minimum_required(VERSION 3.16)
project(libai VERSION 1.0.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(ACL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../third-party/acl" CACHE PATH "Prebuilt ACL root")
set(OPENSSL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../third-party/openssl" CACHE PATH "Prebuilt OpenSSL root")
include(cmake/Dependencies.cmake)
set(AI_MODULES agent provider storage project workspace sandbox context prompt validation common runtime)
set(AI_SOURCES)
foreach(module IN LISTS AI_MODULES)
 file(GLOB module_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${module}/*.cpp")
 list(APPEND AI_SOURCES ${module_sources})
endforeach()
add_library(webcool_ai_coding STATIC ${AI_SOURCES})
option(AI_CODING_USE_PCH "Use precompiled headers for libai" ON)
if(AI_CODING_USE_PCH)
 target_precompile_headers(webcool_ai_coding PRIVATE stdafx.h)
endif()
add_library(libai::ai ALIAS webcool_ai_coding)
# Compatibility for existing source integrations.
add_library(webcool::ai_coding ALIAS webcool_ai_coding)
set_target_properties(webcool_ai_coding PROPERTIES EXPORT_NAME ai OUTPUT_NAME ai POSITION_INDEPENDENT_CODE ON)
target_compile_features(webcool_ai_coding PUBLIC cxx_std_11)
target_include_directories(webcool_ai_coding PUBLIC
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/..>
 $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/libai>
 $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(webcool_ai_coding PUBLIC ACL_USE_CPP11)
target_compile_definitions(webcool_ai_coding PRIVATE _REENTRANT _POSIX_PTHREAD_SEMANTICS _USE_FAST_MACRO)
target_link_libraries(webcool_ai_coding PUBLIC webcool_ai_dependencies)
if(APPLE)
 target_compile_definitions(webcool_ai_coding PRIVATE MACOSX)
 target_compile_options(webcool_ai_coding PRIVATE -Wno-invalid-source-encoding -Wno-invalid-offsetof)
elseif(WIN32)
 set_target_properties(webcool_ai_coding PROPERTIES PREFIX "lib")
 target_compile_definitions(webcool_ai_coding PUBLIC ACL_WINDOWS WIN32_LEAN_AND_MEAN NOMINMAX)
 target_compile_definitions(webcool_ai_coding PRIVATE _CRT_SECURE_NO_WARNINGS _WINSOCK_DEPRECATED_NO_WARNINGS)
elseif(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
 target_compile_definitions(webcool_ai_coding PRIVATE FREEBSD)
endif()
if(MSVC)
 target_compile_options(webcool_ai_coding PRIVATE /utf-8 /Zc:__cplusplus)
endif()
install(TARGETS webcool_ai_coding EXPORT libaiTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/libai FILES_MATCHING PATTERN "*.h" PATTERN "build" EXCLUDE PATTERN "lib" EXCLUDE PATTERN "obj" EXCLUDE PATTERN "tests" EXCLUDE PATTERN "cmake" EXCLUDE)
configure_package_config_file(cmake/libaiConfig.cmake.in
 "${CMAKE_CURRENT_BINARY_DIR}/libaiConfig.cmake"
 INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libai)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/libaiConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT libaiTargets NAMESPACE libai:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libai)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/libaiConfig.cmake"
 "${CMAKE_CURRENT_BINARY_DIR}/libaiConfigVersion.cmake"
 cmake/Dependencies.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libai)

option(AI_CODING_BUILD_TESTS "Build standalone library tests" OFF)
if(AI_CODING_BUILD_TESTS)
 enable_testing()
 add_executable(ai_common_primitives_test tests/common_primitives_test.cpp)
 target_link_libraries(ai_common_primitives_test PRIVATE libai::ai)
 if(MSVC)
  target_compile_options(ai_common_primitives_test PRIVATE /UNDEBUG)
 else()
  target_compile_options(ai_common_primitives_test PRIVATE -UNDEBUG)
 endif()
 add_test(NAME ai_common_primitives COMMAND ai_common_primitives_test)
 add_executable(ai_repair_scope_test tests/repair_scope_test.cpp)
 target_compile_features(ai_repair_scope_test PRIVATE cxx_std_11)
 if(MSVC)
  target_compile_options(ai_repair_scope_test PRIVATE /UNDEBUG /utf-8)
 else()
  target_compile_options(ai_repair_scope_test PRIVATE -UNDEBUG)
 endif()
 add_test(NAME ai_repair_scope COMMAND ai_repair_scope_test)
 add_executable(ai_coding_link_test tests/coding_link_test.cpp)
 target_link_libraries(ai_coding_link_test PRIVATE webcool::ai_coding)
 if(APPLE)
  target_link_options(ai_coding_link_test PRIVATE "-Wl,-force_load,$<TARGET_FILE:webcool_ai_coding>")
 elseif(MSVC)
  target_link_options(ai_coding_link_test PRIVATE "/WHOLEARCHIVE:$<TARGET_FILE:webcool_ai_coding>")
 else()
  target_link_libraries(ai_coding_link_test PRIVATE "-Wl,--whole-archive" webcool_ai_coding "-Wl,--no-whole-archive")
 endif()
 add_test(NAME ai_coding_full_archive_link COMMAND ai_coding_link_test)
endif()
