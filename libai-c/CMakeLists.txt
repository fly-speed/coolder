cmake_minimum_required(VERSION 3.16)
project(libai_c VERSION 1.0.0 LANGUAGES C CXX)
include(GNUInstallDirs)
option(BUILD_SHARED_LIBS "Build shared C ABI" ON)
option(LIBAI_C_TESTS "Build portable bridge tests" ON)
option(LIBAI_CODING_FULL "Enable desktop coding service (ACL/OpenSSL required)" OFF)
find_package(Threads REQUIRED)
add_library(libai_c src/rag.cpp src/workflow.cpp src/mcp.cpp src/file_execution_store.cpp src/bridge.cpp src/builders.cpp src/json_writer.cpp src/coding.cpp src/cancellation.cpp src/provider.cpp )
add_library(libai::c ALIAS libai_c)
set_target_properties(libai_c PROPERTIES POSITION_INDEPENDENT_CODE ON OUTPUT_NAME ai_c EXPORT_NAME c VERSION 1.0.0 SOVERSION 1)
target_compile_features(libai_c PUBLIC cxx_std_11)
target_include_directories(libai_c PUBLIC
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
 $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_include_directories(libai_c PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_link_libraries(libai_c PRIVATE Threads::Threads)
target_compile_definitions(libai_c PRIVATE LIBAI_C_BUILD)
if(BUILD_SHARED_LIBS)
 target_compile_definitions(libai_c PUBLIC LIBAI_C_SHARED)
endif()
install(TARGETS libai_c EXPORT libai_cTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
 LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/libai.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES README.md ../libai/docs/MCP.md ../libai/docs/WORKFLOW.md ../libai/docs/RAG.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
if(LIBAI_C_TESTS)
 enable_testing()
 add_executable(libai_c_deadline_test tests/deadline_test.c)
 target_link_libraries(libai_c_deadline_test PRIVATE libai_c)
 target_compile_options(libai_c_deadline_test PRIVATE $<$<C_COMPILER_ID:MSVC>:/UNDEBUG> $<$<NOT:$<C_COMPILER_ID:MSVC>>:-UNDEBUG>)
 add_test(NAME libai_c_deadline COMMAND libai_c_deadline_test)
 add_executable(libai_c_rag_test tests/rag_test.c)
 target_link_libraries(libai_c_rag_test PRIVATE libai_c)
 target_compile_options(libai_c_rag_test PRIVATE $<$<C_COMPILER_ID:MSVC>:/UNDEBUG> $<$<NOT:$<C_COMPILER_ID:MSVC>>:-UNDEBUG>)
 add_test(NAME libai_c_rag COMMAND libai_c_rag_test)
 set_tests_properties(libai_c_rag PROPERTIES TIMEOUT 30)
 add_executable(libai_c_workflow_test tests/workflow_test.c)
 target_link_libraries(libai_c_workflow_test PRIVATE libai_c Threads::Threads)
 target_compile_options(libai_c_workflow_test PRIVATE $<$<C_COMPILER_ID:MSVC>:/UNDEBUG> $<$<NOT:$<C_COMPILER_ID:MSVC>>:-UNDEBUG>)
 add_test(NAME libai_c_workflow COMMAND libai_c_workflow_test)
 set_tests_properties(libai_c_workflow PROPERTIES TIMEOUT 20)
 add_executable(libai_c_mcp_test tests/mcp_test.c)
 target_link_libraries(libai_c_mcp_test PRIVATE libai_c)
 target_compile_options(libai_c_mcp_test PRIVATE $<$<C_COMPILER_ID:MSVC>:/UNDEBUG> $<$<NOT:$<C_COMPILER_ID:MSVC>>:-UNDEBUG>)
 add_test(NAME libai_c_mcp COMMAND libai_c_mcp_test)
 set_tests_properties(libai_c_mcp PROPERTIES TIMEOUT 15)
 if(UNIX AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_FOUND)
   add_test(NAME libai_c_mcp_stdio COMMAND libai_c_mcp_test "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../libai-c++/tests/mcp_stdio_fixture.py")
   set_tests_properties(libai_c_mcp_stdio PROPERTIES TIMEOUT 15)
  endif()
 endif()
 add_executable(libai_c_output_test tests/structured_output_test.c)
 target_link_libraries(libai_c_output_test PRIVATE libai_c)
 target_compile_options(libai_c_output_test PRIVATE $<$<C_COMPILER_ID:MSVC>:/UNDEBUG> $<$<NOT:$<C_COMPILER_ID:MSVC>>:-UNDEBUG>)
 add_test(NAME libai_c_output COMMAND libai_c_output_test)
 add_executable(libai_c_test tests/bridge_test.c)
 target_link_libraries(libai_c_test PRIVATE libai_c)
 if(MSVC)
  target_compile_options(libai_c_test PRIVATE /UNDEBUG)
 else()
  target_compile_options(libai_c_test PRIVATE -UNDEBUG)
 endif()
 add_test(NAME libai_c_bridge COMMAND libai_c_test)
endif()

include(CMakePackageConfigHelpers)
configure_package_config_file(libai_cConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/libai_cConfig.cmake"
 INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/libai_c")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/libai_cConfigVersion.cmake"
 VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT libai_cTargets NAMESPACE libai:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/libai_c")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/libai_cConfig.cmake"
 "${CMAKE_CURRENT_BINARY_DIR}/libai_cConfigVersion.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/libai_c")

if(LIBAI_CODING_FULL)
 include("${CMAKE_CURRENT_SOURCE_DIR}/../libai/cmake/CodingBackend.cmake")
 libai_enable_coding(libai_c)
 install(FILES ../libai/cmake/Dependencies.cmake DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/libai_c")
endif()
