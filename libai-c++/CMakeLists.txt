cmake_minimum_required(VERSION 3.16)
project(libai_cpp VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Threads REQUIRED)
option(BUILD_SHARED_LIBS "Build a shared C++ application library (default: static)" OFF)
option(LIBAI_CPP_BUILD_TESTS "Build C++ facade integration tests" OFF)
option(LIBAI_CPP_BUILD_EXAMPLES "Build a standalone application example" ON)
option(LIBAI_CPP_FULL "Include desktop coding facilities (ACL/OpenSSL required)" OFF)
add_library(libai_cpp src/rag.cpp src/workflow.cpp src/mcp.cpp src/file_execution_store.cpp src/agent.cpp src/coding_service.cpp )
add_library(libai::cpp ALIAS libai_cpp)
set_target_properties(libai_cpp PROPERTIES EXPORT_NAME cpp OUTPUT_NAME ai_cpp
 VERSION ${PROJECT_VERSION} SOVERSION 1 POSITION_INDEPENDENT_CODE ON)
target_compile_features(libai_cpp PUBLIC cxx_std_11)
target_include_directories(libai_cpp PUBLIC
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
 $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(libai_cpp PRIVATE Threads::Threads)
target_include_directories(libai_cpp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_definitions(libai_cpp PRIVATE LIBAI_CPP_BUILD)
if(BUILD_SHARED_LIBS)
 target_compile_definitions(libai_cpp PUBLIC LIBAI_CPP_SHARED)
endif()
if(LIBAI_CPP_FULL AND WIN32 AND BUILD_SHARED_LIBS)
 message(FATAL_ERROR "Full typed Windows facade requires BUILD_SHARED_LIBS=OFF; C/JNI coding bindings can be shared")
endif()
if(LIBAI_CPP_FULL)
 include(cmake/CppCodingBackend.cmake)
endif()
install(TARGETS libai_cpp EXPORT libai_cppTargets
 ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
 RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
configure_package_config_file(cmake/libai_cppConfig.cmake.in
 "${CMAKE_CURRENT_BINARY_DIR}/libai_cppConfig.cmake"
 INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libai_cpp)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/libai_cppConfigVersion.cmake"
 VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT libai_cppTargets NAMESPACE libai:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libai_cpp)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/libai_cppConfig.cmake"
 "${CMAKE_CURRENT_BINARY_DIR}/libai_cppConfigVersion.cmake"
 DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libai_cpp)
if(LIBAI_CPP_FULL)
 install(FILES ../libai/cmake/Dependencies.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/libai_cpp)
endif()
install(FILES README.md ../LICENSE ../libai/docs/RAG.md ../libai/docs/WORKFLOW.md ../libai/docs/MCP.md ../libai/docs/CODING_SERVICE.md ../libai/docs/AGENT_PLATFORM_PROGRESS.md ../libai/docs/GOVERNANCE.md ../libai/docs/DURABLE_EXECUTION.md ../libai/docs/STRUCTURED_OUTPUT.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/libai_cpp/examples)
if(LIBAI_CPP_BUILD_EXAMPLES)
 add_subdirectory(examples)
endif()
