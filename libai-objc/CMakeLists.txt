cmake_minimum_required(VERSION 3.16)
project(libai_objc LANGUAGES C CXX OBJCXX)
if(NOT APPLE)
 message(FATAL_ERROR "Objective-C binding requires an Apple Foundation SDK")
endif()
option(LIBAI_OBJC_TESTS "Build Foundation binding tests" ON)
find_package(Threads REQUIRED)
add_library(libai_objc STATIC src/LAIAgent.mm src/LAIValues.mm src/LAICallbacks.mm src/LAICodingService.mm
 ../libai-c/src/bridge.cpp ../libai-c/src/builders.cpp ../libai-c/src/json_writer.cpp ../libai-c/src/coding.cpp ../libai-c/src/cancellation.cpp ../libai-c/src/provider.cpp)
set_target_properties(libai_objc PROPERTIES POSITION_INDEPENDENT_CODE ON OUTPUT_NAME ai_objc)
target_compile_features(libai_objc PRIVATE cxx_std_11)
target_include_directories(libai_objc PUBLIC include PRIVATE ../libai-c/include ..)
target_compile_options(libai_objc PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>" "$<$<COMPILE_LANGUAGE:OBJCXX>:-fblocks>")
target_link_libraries(libai_objc PUBLIC "-framework Foundation" Threads::Threads)
install(TARGETS libai_objc ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include/LibAI)
if(LIBAI_OBJC_TESTS AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
 enable_testing()
 add_executable(libai_objc_test tests/agent_test.mm)
 target_compile_options(libai_objc_test PRIVATE -fobjc-arc -fblocks -UNDEBUG)
 target_link_libraries(libai_objc_test PRIVATE libai_objc)
 add_test(NAME libai_objc_bridge COMMAND libai_objc_test)
endif()

option(LIBAI_CODING_FULL "Enable desktop coding service (ACL/OpenSSL required)" OFF)
if(LIBAI_CODING_FULL)
 include("${CMAKE_CURRENT_SOURCE_DIR}/../libai/cmake/CodingBackend.cmake")
 libai_enable_coding(libai_objc)
endif()

if(LIBAI_OBJC_TESTS AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
 file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/coding-test-root")
 add_executable(libai_objc_coding_test tests/coding_test.mm)
 target_compile_options(libai_objc_coding_test PRIVATE -fobjc-arc -UNDEBUG)
 target_link_libraries(libai_objc_coding_test PRIVATE libai_objc)
 add_test(NAME libai_objc_coding COMMAND libai_objc_coding_test "${CMAKE_CURRENT_BINARY_DIR}/coding-test-root")
endif()
